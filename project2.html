Ingestion, retrieval, and the eval loop

ingest.py · retrieval_agent.py

Two pipelines share one vector store: an offline ingestion job that chunks and embeds the docs, and a runtime agent that decides when to call retrieval as a tool. The eval harness sits outside both, replaying a golden set through the live agent on every change.

  • Vector store — one collection, shared by ingestion and runtime. Chunks carry source + page metadata for citations.
  • Docs Agent — sees the question and its own seen-docs memory before deciding whether retrieval is even needed. Not every question needs the knowledge base.
  • Seen-docs memory — tracks which chunks were already surfaced this session, so a follow-up doesn't re-retrieve and re-pay for the same context.
  • Eval harness — runs independently of the chat flow. Replays the golden set through the live agent and scores each answer with an LLM judge for faithfulness and relevance.

ingest.py — offline chunk + embed pipeline
"""
ingest.py — Chunks source docs and writes embeddings to the shared vector
store. Run whenever the doc set changes: python ingest.py ./docs
"""
import sys, hashlib
from pathlib import Path
import chromadb
from chromadb.utils import embedding_functions

CHUNK_SIZE = 512
CHUNK_OVERLAP = 64

client = chromadb.PersistentClient(path="./vector_store")
embed_fn = embedding_functions.DefaultEmbeddingFunction()
collection = client.get_or_create_collection("docs", embedding_function=embed_fn)

def chunk_text(text: str, source: str) -> list[dict]:
    # Recursive splitting on paragraph -> sentence boundaries, then a
    # fixed-size fallback, so chunks don't cut mid-sentence.
    paragraphs = [p for p in text.split("\n\n") if p.strip()]
    chunks, buf = [], ""
    for para in paragraphs:
        if len(buf) + len(para) > CHUNK_SIZE and buf:
            chunks.append(buf.strip())
            buf = buf[-CHUNK_OVERLAP:]  # small overlap for continuity
        buf += "\n\n" + para
    if buf.strip():
        chunks.append(buf.strip())
    return [{"text": c, "source": source} for c in chunks]

def ingest_directory(path: str) -> None:
    for file in Path(path).rglob("*.md"):
        text = file.read_text(encoding="utf-8")
        for chunk in chunk_text(text, source=str(file)):
            cid = hashlib.sha256(chunk["text"].encode()).hexdigest()[:16]
            collection.upsert(
                ids=[cid],
                documents=[chunk["text"]],
                metadatas=[{"source": chunk["source"]}],
            )
    print(f"Ingested into collection: {collection.count()} chunks total")

if __name__ == "__main__":
    ingest_directory(sys.argv[1] if len(sys.argv) > 1 else "./docs")
retrieval_agent.py — agent that decides when to retrieve
"""
retrieval_agent.py — Retrieval is exposed as a tool, not a hardcoded
pre-step. The agent reads the question and its own seen-docs memory,
then decides whether retrieval is needed at all.
"""
from anthropic import Anthropic
import chromadb
from memory import SeenDocsMemory

client = Anthropic()
store = chromadb.PersistentClient(path="./vector_store").get_collection("docs")
memory = SeenDocsMemory()

SYSTEM = """You are a support docs agent. You have a retrieve_docs tool over \
the product documentation. Only call it when the question needs specific \
facts you don't already have from earlier in this session — for general \
or already-answered questions, answer directly. Always cite sources by \
file name when you use retrieved text."""

RETRIEVE_TOOL = {
    "name": "retrieve_docs",
    "description": "Search the product documentation for relevant passages.",
    "input_schema": {
        "type": "object",
        "properties": {
            "query": {"type": "string"},
            "top_k": {"type": "integer", "default": 4},
        },
        "required": ["query"],
    },
}

def retrieve_docs(query: str, top_k: int = 4, session_id: str = "default") -> list[dict]:
    results = store.query(query_texts=[query], n_results=top_k)
    hits = [
        {"text": doc, "source": meta["source"], "score": 1 - dist}
        for doc, meta, dist in zip(
            results["documents"][0], results["metadatas"][0], results["distances"][0]
        )
    ]
    # Drop chunks already surfaced this session before returning to the agent
    fresh = memory.filter_unseen(session_id, hits)
    memory.mark_seen(session_id, fresh)
    return fresh

def answer(question: str, session_id: str = "default") -> str:
    response = client.messages.create(
        model="claude-sonnet-4-6",
        max_tokens=800,
        system=[{"type": "text", "text": SYSTEM,
                 "cache_control": {"type": "ephemeral"}}],
        tools=[RETRIEVE_TOOL],
        messages=[{"role": "user", "content": question}],
    )

    for block in response.content:
        if block.type == "tool_use" and block.name == "retrieve_docs":
            hits = retrieve_docs(session_id=session_id, **block.input)
            return continue_with_tool_result(response, block, hits)

    return response.content[0].text  # answered without retrieving
memory.py — lightweight seen-docs memory
"""
memory.py — Tracks which chunks have already been surfaced in a session
so a follow-up question doesn't re-retrieve (and re-pay for) the same
context. Deliberately simple: an in-memory set per session, not a full
conversation store.
"""
from collections import defaultdict

class SeenDocsMemory:
    def __init__(self):
        self._seen: dict[str, set[str]] = defaultdict(set)

    def filter_unseen(self, session_id: str, hits: list[dict]) -> list[dict]:
        seen = self._seen[session_id]
        unseen = [h for h in hits if h["source"] not in seen]
        # If everything was already seen, fall back to the top hit anyway —
        # better to re-cite than to answer with nothing.
        return unseen or hits[:1]

    def mark_seen(self, session_id: str, hits: list[dict]) -> None:
        for h in hits:
            self._seen[session_id].add(h["source"])

    def summary(self, session_id: str) -> str:
        seen = self._seen[session_id]
        return f"{len(seen)} source(s) already surfaced: " + ", ".join(sorted(seen)) if seen else "none yet"
eval_harness.py — golden set + LLM-as-judge, wired for CI
"""
eval_harness.py — Replays a golden Q&A set through the live agent and
scores each answer for faithfulness and relevance with an LLM judge.
Run in CI on every prompt or chunking change: pytest eval_harness.py
"""
import json, statistics
from anthropic import Anthropic
from retrieval_agent import answer

client = Anthropic()
GOLDEN_SET_PATH = "eval/golden_set.json"
FAITHFULNESS_THRESHOLD = 0.85
RELEVANCE_THRESHOLD = 0.80

JUDGE_PROMPT = """Score the ANSWER against the SOURCE CONTEXT and the QUESTION.
Return strict JSON: {{"faithfulness": 0-1, "relevance": 0-1, "reason": str}}
faithfulness = is every claim in the answer supported by the context?
relevance = does the answer actually address the question asked?

QUESTION: {question}
CONTEXT: {context}
ANSWER: {answer}"""

def judge(question: str, context: str, answer_text: str) -> dict:
    resp = client.messages.create(
        model="claude-sonnet-4-6",
        max_tokens=200,
        messages=[{"role": "user",
                   "content": JUDGE_PROMPT.format(
                       question=question, context=context, answer=answer_text)}],
    )
    return json.loads(resp.content[0].text)

def run_regression() -> dict:
    golden_set = json.load(open(GOLDEN_SET_PATH))
    results = []
    for case in golden_set:
        response = answer(case["question"], session_id=f"eval-{case['id']}")
        scores = judge(case["question"], case["expected_context"], response)
        passed = (
            scores["faithfulness"] >= FAITHFULNESS_THRESHOLD
            and scores["relevance"] >= RELEVANCE_THRESHOLD
        )
        results.append({"id": case["id"], "passed": passed, **scores})

    pass_rate = sum(r["passed"] for r in results) / len(results)
    avg_faith = statistics.mean(r["faithfulness"] for r in results)
    return {"pass_rate": pass_rate, "avg_faithfulness": avg_faith, "cases": results}

def test_regression_gate():
    # CI fails the build if pass rate drops below the committed baseline
    report = run_regression()
    assert report["pass_rate"] >= 0.90, f"Eval pass rate dropped to {report['pass_rate']:.0%}"
