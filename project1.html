PROJECT 01 / MULTI-AGENT SYSTEM

A research pipeline that plans, delegates, and cites its sources.

Four agents — Orchestrator, Researcher, Analyst, Writer — cooperate to turn an open question into a sourced report. The Researcher reaches into a custom MCP server for the client's private knowledge base, session memory carries context across follow-up questions, and prompt caching keeps repeat calls cheap.

How the pipeline is wired

orchestrator.py

The Orchestrator never talks to tools directly — it plans sub-questions and hands them to the Researcher, which is the only agent with tool access. This keeps the tool surface small and auditable.

  • Orchestrator — turns the user's question into 2–4 concrete sub-questions, informed by the session's memory summary.
  • Researcher — the only agent with tool access. Chooses between public web search and the MCP-connected knowledge base per sub-question.
  • MCP server — a standalone process exposing search_documents over the Model Context Protocol, so any MCP-compatible host can reuse it, not just this pipeline.
  • Analyst — merges findings, flags contradictions between sources, drops low-confidence claims.
  • Writer — drafts the final report with inline citations back to source snippets.

Why context engineering matters here

cache_control: ephemeral

Each agent's system prompt — role instructions, tool schemas, output format — is identical on every call within a session. Splitting it from the per-turn dynamic context (the memory summary, the current sub-question) lets that static block get cached, so only the small dynamic slice is billed at full price on repeat calls.

~2,100
tokens of static system prompt + tool schemas, cached after the first call
~180
tokens of dynamic context per turn — memory summary + current sub-question
3.1×
cheaper per turn from the 2nd call onward in the same session

orchestrator.py — wires the four agents into a stateful graph
"""
orchestrator.py — Wires the four agents into a stateful graph and owns
the run loop, including session memory persistence.
"""
from langgraph.graph import StateGraph, END
from typing import TypedDict
from agents import researcher_turn, analyst_turn, writer_turn, plan_subtasks
from memory import SessionMemory

class PipelineState(TypedDict):
    query: str
    subtasks: list[str]
    findings: list[dict]
    analysis: str
    report: str
    memory_summary: str

memory = SessionMemory(store_path="sessions.db")

def orchestrate(state: PipelineState) -> PipelineState:
    # Plan 2-4 sub-questions, informed by prior session context
    state["subtasks"] = plan_subtasks(state["query"], state["memory_summary"])
    return state

def research(state: PipelineState) -> PipelineState:
    state["findings"] = [
        researcher_turn(q, state["memory_summary"]) for q in state["subtasks"]
    ]
    return state

def analyze(state: PipelineState) -> PipelineState:
    state["analysis"] = analyst_turn(state["findings"])
    return state

def write(state: PipelineState) -> PipelineState:
    state["report"] = writer_turn(state["analysis"], state["query"])
    return state

graph = StateGraph(PipelineState)
graph.add_node("orchestrate", orchestrate)
graph.add_node("research", research)
graph.add_node("analyze", analyze)
graph.add_node("write", write)
graph.set_entry_point("orchestrate")
graph.add_edge("orchestrate", "research")
graph.add_edge("research", "analyze")
graph.add_edge("analyze", "write")
graph.add_edge("write", END)

pipeline = graph.compile()

def run(query: str, session_id: str) -> str:
    prior = memory.load_summary(session_id)
    result = pipeline.invoke({
        "query": query, "memory_summary": prior,
        "subtasks": [], "findings": [], "analysis": "", "report": "",
    })
    memory.save_turn(session_id, query, result["report"])
    return result["report"]
mcp_server.py — exposes the client's knowledge base as an MCP tool
"""
mcp_server.py — Custom MCP server exposing the client's document/knowledge
store as a tool any MCP-compatible agent host can call, not just this
pipeline. Run standalone: python mcp_server.py
"""
from mcp.server import Server
from mcp.server.stdio import stdio_server
from mcp.types import Tool, TextContent
import json
from knowledge_store import DocumentStore  # thin wrapper over a vector DB

store = DocumentStore(collection="client_knowledge_base")
server = Server("research-knowledge-mcp")

@server.list_tools()
async def list_tools() -> list[Tool]:
    return [
        Tool(
            name="search_documents",
            description=(
                "Search the client's internal knowledge base (reports, "
                "market data, prior research notes) for passages relevant "
                "to a query. Returns ranked snippets with source metadata."
            ),
            inputSchema={
                "type": "object",
                "properties": {
                    "query": {"type": "string"},
                    "top_k": {"type": "integer", "default": 5},
                },
                "required": ["query"],
            },
        )
    ]

@server.call_tool()
async def call_tool(name: str, arguments: dict) -> list[TextContent]:
    if name != "search_documents":
        raise ValueError(f"Unknown tool: {name}")

    results = store.similarity_search(
        arguments["query"], k=arguments.get("top_k", 5)
    )
    payload = [
        {"source": r.source, "score": round(r.score, 3), "text": r.text}
        for r in results
    ]
    return [TextContent(type="text", text=json.dumps(payload, indent=2))]

if __name__ == "__main__":
    import asyncio
    asyncio.run(stdio_server(server))
agents.py — Researcher agent with tool calling + prompt caching
"""
agents.py — Each agent is a thin wrapper around a system prompt, a
toolset, and a call to Claude with prompt caching on the static portion
of context. Researcher shown in full; Analyst/Writer follow the same
shape with no tools attached.
"""
from anthropic import Anthropic
from mcp_client import MCPToolClient  # connects to mcp_server.py over stdio

client = Anthropic()
mcp = MCPToolClient(server_command=["python", "mcp_server.py"])

RESEARCHER_SYSTEM = """You are the Researcher agent in a multi-agent \
research pipeline. Given a sub-question, decide whether to search the web, \
the internal knowledge base (via MCP), or both. Always cite sources. \
Return findings as JSON: [{claim, source, confidence}]."""

WEB_SEARCH_TOOL = {
    "name": "web_search",
    "description": "Search the public web for recent, dated information.",
    "input_schema": {
        "type": "object",
        "properties": {"query": {"type": "string"}},
        "required": ["query"],
    },
}

def researcher_turn(sub_question: str, memory_summary: str) -> dict:
    tools = [WEB_SEARCH_TOOL, *mcp.as_anthropic_tools()]

    response = client.messages.create(
        model="claude-sonnet-4-6",
        max_tokens=1500,
        system=[
            # Static instructions: cached, reused across every sub-question
            {"type": "text", "text": RESEARCHER_SYSTEM,
             "cache_control": {"type": "ephemeral"}},
            # Dynamic context: not cached, changes every turn
            {"type": "text", "text": f"Session memory:\n{memory_summary}"},
        ],
        tools=tools,
        messages=[{"role": "user", "content": sub_question}],
    )

    for block in response.content:
        if block.type == "tool_use":
            if block.name == "web_search":
                result = run_web_search(block.input["query"])
            else:
                result = mcp.call(block.name, block.input)
            return continue_with_tool_result(response, block, result)

    return {"text": response.content[0].text}
memory.py — bounded, summarized memory across sessions
"""
memory.py — Cross-session memory. Keeps a rolling summary per session so
follow-up questions don't restart the pipeline from zero context, without
replaying full transcripts (which would blow the cached-prefix and cost).
"""
import sqlite3, json
from anthropic import Anthropic

client = Anthropic()

class SessionMemory:
    def __init__(self, store_path: str):
        self.db = sqlite3.connect(store_path)
        self.db.execute(
            "CREATE TABLE IF NOT EXISTS sessions "
            "(session_id TEXT PRIMARY KEY, summary TEXT, turns TEXT)"
        )

    def load_summary(self, session_id: str) -> str:
        row = self.db.execute(
            "SELECT summary FROM sessions WHERE session_id = ?", (session_id,)
        ).fetchone()
        return row[0] if row else "No prior context."

    def save_turn(self, session_id: str, query: str, report: str) -> None:
        prior = self.load_summary(session_id)
        # Re-summarize rather than append verbatim — keeps memory bounded
        # and keeps the cached system-prompt prefix stable across turns.
        summary = self._resummarize(prior, query, report)
        self.db.execute(
            "INSERT INTO sessions (session_id, summary, turns) VALUES (?,?,?) "
            "ON CONFLICT(session_id) DO UPDATE SET summary=excluded.summary",
            (session_id, summary, json.dumps({"query": query})),
        )
        self.db.commit()

    def _resummarize(self, prior: str, query: str, report: str) -> str:
        resp = client.messages.create(
            model="claude-sonnet-4-6",
            max_tokens=300,
            messages=[{
                "role": "user",
                "content": (
                    f"Prior session summary:\n{prior}\n\n"
                    f"New question: {query}\nNew report excerpt: {report[:800]}\n\n"
                    "Write an updated 3-4 sentence summary of what this "
                    "session has covered, for use as context in future turns."
                ),
            }],
        )
        return resp.content[0].text
