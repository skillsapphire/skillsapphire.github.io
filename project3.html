An agent that proposes, never just acts.

A single ticket-triage agent proposes an action — a refund, an escalation, a reply — but nothing executes until it clears a schema/allowlist guardrail and a human approval gate. Every step is traced: latency, tokens, and pass/fail, feeding a CI suite that red-teams the guardrails on every prompt change.

The agent's tool call doesn't execute anything directly — it returns a structured, schema-validated proposal. That proposal has to pass an allowlist check and a human approval before an execution layer touches anything, and every hop emits a trace span.

  • Ops Agent — triages the ticket and calls a propose_action tool. It never has a tool that executes anything directly.
  • Guardrail — validates the proposal against a JSON schema and a hard allowlist of permitted actions and limits (e.g. max refund amount).
  • Approval gate — every proposal that passes the guardrail still waits for a human decision before execution. No auto-approve path exists in this design.
  • Execute — the only component with real side effects, and it only runs after both the guardrail and the human have signed off.
  • Observability — every stage emits a span with latency, token usage, and pass/fail, exported via OpenTelemetry.

workflow_agent.py — triages and proposes, never executes directly
"""
workflow_agent.py — Triages a ticket and calls propose_action, a tool
that returns a structured proposal. The agent has no tool capable of
actually changing anything — that boundary is deliberate.
"""
from anthropic import Anthropic
from guardrails import validate_proposal
from approval_gate import request_approval
from observability import traced_span
import executor

client = Anthropic()

SYSTEM = """You triage support tickets and propose one action: issue_refund, \
escalate, or reply_only. Never claim an action has happened — you can \
only propose it. Stay within policy limits described in the tool schema."""

PROPOSE_ACTION_TOOL = {
    "name": "propose_action",
    "description": "Propose one action for a human to review. Does not execute anything.",
    "input_schema": {
        "type": "object",
        "properties": {
            "action": {"type": "string", "enum": ["issue_refund", "escalate", "reply_only"]},
            "amount_usd": {"type": "number"},
            "reason": {"type": "string"},
        },
        "required": ["action", "reason"],
    },
}

@traced_span("agent.triage")
def triage(ticket_text: str) -> dict:
    response = client.messages.create(
        model="claude-sonnet-4-6",
        max_tokens=500,
        system=[{"type": "text", "text": SYSTEM,
                 "cache_control": {"type": "ephemeral"}}],
        tools=[PROPOSE_ACTION_TOOL],
        messages=[{"role": "user", "content": ticket_text}],
    )
    proposal = next(b.input for b in response.content if b.type == "tool_use")
    return proposal

def handle_ticket(ticket_text: str) -> str:
    proposal = triage(ticket_text)

    check = validate_proposal(proposal)  # schema + allowlist, see guardrails.py
    if not check.passed:
        return f"Blocked by guardrail: {check.reason}"

    decision = request_approval(proposal)  # blocks on a human, see approval_gate.py
    if not decision.approved:
        return f"Rejected by {decision.approver}: {decision.note}"

    return executor.run(proposal, approved_by=decision.approver)
guardrails.py — schema validation + hard action allowlist
"""
guardrails.py — Two independent checks before anything reaches a human:
a JSON-schema validation of the proposal shape, and a hard allowlist of
permitted actions with policy limits (e.g. max refund amount).
"""
from dataclasses import dataclass
from jsonschema import validate, ValidationError

ALLOWED_ACTIONS = {"issue_refund", "escalate", "reply_only"}
MAX_REFUND_USD = 250.0
MAX_REFUND_REQUIRING_SECOND_APPROVER = 250.0

PROPOSAL_SCHEMA = {
    "type": "object",
    "properties": {
        "action": {"type": "string"},
        "amount_usd": {"type": "number"},
        "reason": {"type": "string", "minLength": 5},
    },
    "required": ["action", "reason"],
}

@dataclass
class GuardrailResult:
    passed: bool
    reason: str = ""
    requires_second_approver: bool = False

def validate_proposal(proposal: dict) -> GuardrailResult:
    try:
        validate(proposal, PROPOSAL_SCHEMA)
    except ValidationError as e:
        return GuardrailResult(passed=False, reason=f"schema: {e.message}")

    if proposal["action"] not in ALLOWED_ACTIONS:
        return GuardrailResult(passed=False, reason=f"'{proposal['action']}' not in allowlist")

    if proposal["action"] == "issue_refund":
        amount = proposal.get("amount_usd", 0)
        if amount > MAX_REFUND_USD:
            return GuardrailResult(passed=False, reason=f"${amount} exceeds ${MAX_REFUND_USD} auto-propose limit")
        return GuardrailResult(
            passed=True,
            requires_second_approver=amount >= MAX_REFUND_REQUIRING_SECOND_APPROVER,
        )

    return GuardrailResult(passed=True)
approval_gate.py — human-in-the-loop, no auto-approve path
"""
approval_gate.py — Every guardrail-passed proposal still blocks on a
human decision. This CLI version is a stand-in for the real Slack
approval flow used in the deployed pipeline.
"""
from dataclasses import dataclass
from observability import traced_span

@dataclass
class ApprovalDecision:
    approved: bool
    approver: str
    note: str = ""

@traced_span("approval.request")
def request_approval(proposal: dict) -> ApprovalDecision:
    # In production this posts to Slack with Approve/Reject buttons and
    # blocks on a webhook callback instead of stdin. Same contract either way:
    # nothing downstream runs until an ApprovalDecision comes back.
    print(f"\nProposal: {proposal}\nApprove? [y/N]: ", end="")
    reply = input().strip().lower()

    if reply == "y":
        return ApprovalDecision(approved=True, approver="cli-operator")
    return ApprovalDecision(approved=False, approver="cli-operator", note="declined at approval gate")
observability.py — OpenTelemetry span wrapper for every stage
"""
observability.py — A decorator that wraps any pipeline stage in an
OpenTelemetry span, capturing latency and success/failure. Token usage
is attached separately by the caller when the wrapped call is an LLM call.
"""
import functools, time
from opentelemetry import trace

tracer = trace.get_tracer("ticket-ops-agent")

def traced_span(span_name: str):
    def decorator(fn):
        @functools.wraps(fn)
        def wrapper(*args, **kwargs):
            with tracer.start_as_current_span(span_name) as span:
                start = time.perf_counter()
                try:
                    result = fn(*args, **kwargs)
                    span.set_attribute("status", "ok")
                    return result
                except Exception as e:
                    span.set_attribute("status", "error")
                    span.set_attribute("error.message", str(e))
                    raise
                finally:
                    span.set_attribute("duration_ms", round((time.perf_counter() - start) * 1000, 1))
        return wrapper
    return decorator

def record_token_usage(span_name: str, input_tokens: int, output_tokens: int) -> None:
    span = trace.get_current_span()
    span.set_attribute("llm.input_tokens", input_tokens)
    span.set_attribute("llm.output_tokens", output_tokens)
.github/workflows/ci.yml — guardrail suite gates every merge
name: agent-ci

on:
  pull_request:
    paths:
      - "agents/**"
      - "guardrails.py"

jobs:
  guardrail-suite:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - run: pip install -r requirements.txt
      - name: Run guardrail red-team suite
        run: pytest test_guardrails.py -v
        env:
          ANTHROPIC_API_KEY: ${{ secrets.ANTHROPIC_API_KEY }}
      - name: Fail build if any adversarial case slips through
        run: |
          python -c "
          import json
          r = json.load(open('guardrail_report.json'))
          assert r['blocked_rate'] == 1.0, f'{r[\"blocked_rate\"]:.0%} of attacks blocked, need 100%'
          "
